### 第1步：用户登录获取 access_token

\\\javascript
// 微信小程序登录
wx.login({
  success: async (res) => {
    const { code } = res;
    
    // 调用后端登录接口
    const loginResult = await fetch('https://your-domain.com/userlogin', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ js_code: code })
    });
    
    const data = await loginResult.json();
    
    if (data.succeed) {
      const accessToken = data.result.access_token;
      const openid = data.result.openid;
      
      // 保存到本地存储
      wx.setStorageSync('access_token', accessToken);
      wx.setStorageSync('openid', openid);
    }
  }
});

### 第4步：报名课程

\\\javascript
// 报名课程
async function enrollCourse(courseId) {
  const accessToken = wx.getStorageSync('access_token');
  const userId = wx.getStorageSync('openid');
  const userName = wx.getStorageSync('userName') || '用户';
  
  const response = await fetch('https://your-domain.com/api/courses/enroll', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      access_token: accessToken,
      courseId: courseId,
      userId: userId,
      userName: userName
    })
  });
  
  const result = await response.json();
  
  if (result.code === 200) {
    // 报名成功，获取报名信息
    const enrollmentData = result.data;
    /*
    {
      enrollmentId: "ENROLL-xxx",
      courseId: "COURSE-xxx",
      courseTitle: "网球正手技术提升训练",
      courseDate: "2026-01-10",
      courseTime: "10:00",
      location: "深圳湾体育中心1号场",
      amount: 150.00,
      paymentStatus: 1  // 1-待支付
    }
    */
    
    // 继续支付流程
    return enrollmentData;
  } else if (result.code === 400 && result.message.includes('待支付')) {
    // 已有待支付记录，可以继续支付
    return result.data;
  } else {
    wx.showToast({ title: result.message, icon: 'none' });
    return null;
  }
}

### 第6步：发起微信支付

\\\javascript
// 发起微信支付
async function payForCourse(orderId, amount, courseTitle, courseId) {
  const accessToken = wx.getStorageSync('access_token');
  const openid = wx.getStorageSync('openid');
  
  // 调用支付接口
  const response = await fetch('https://your-domain.com/payOrder', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      access_token: accessToken,
      order_id: orderId,
      amount: amount,
      description: courseTitle + ' - 课程报名',
      openid: openid,
      act_id: courseId
    })
  });
  
  const result = await response.json();
  
  if (result.code === 200 && result.succeed) {
    // 获取支付参数
    const payParams = result.response;
    /*
    {
      appId: "wxd678efh567hg6787",
      timeStamp: "1735977600",
      nonceStr: "abc123def456",
      package: "prepay_id=wx20261006123456789",
      signType: "RSA",
      paySign: "签名字符串"
    }
    */
    
    // 调起微信支付
    wx.requestPayment({
      timeStamp: payParams.timeStamp,
      nonceStr: payParams.nonceStr,
      package: payParams.package,
      signType: payParams.signType,
      paySign: payParams.paySign,
      success: function(res) {
        console.log('支付成功', res);
        // 调用后端更新报名状态
        updateEnrollmentStatus(orderId, courseId);
      },
      fail: function(err) {
        console.log('支付失败', err);
        wx.showToast({ title: '支付失败', icon: 'none' });
      }
    });
    
    return true;
  } else {
    wx.showToast({ title: result.msg || '支付请求失败', icon: 'none' });
    return false;
  }
}

### 第7步：更新报名状态（支付成功后）

\\\javascript
// 更新报名状态
async function updateEnrollmentStatus(orderId, courseId) {
  const accessToken = wx.getStorageSync('access_token');
  const userId = wx.getStorageSync('openid');
  
  const response = await fetch('https://your-domain.com/api/courses/payment/callback', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      access_token: accessToken,
      order_id: orderId,
      course_id: courseId,
      user_id: userId,
      payment_id: orderId  // 可选
    })
  });
  
  const result = await response.json();
  
  if (result.code === 200) {
    wx.showToast({ title: '报名成功！', icon: 'success' });
    
    // 跳转到成功页面
    wx.navigateTo({
      url: '/pages/course/enrollSuccess?courseId=' + courseId
    });
  } else {
    wx.showToast({ title: result.message, icon: 'none' });
  }
}

### 完整流程整合

\\\javascript
// 完整的课程报名+支付流程
async function enrollAndPay(courseId) {
  try {
    // 1. 报名课程
    wx.showLoading({ title: '正在报名...' });
    const enrollmentData = await enrollCourse(courseId);
    
    if (!enrollmentData) {
      wx.hideLoading();
      return;
    }
    
    // 2. 生成订单
    wx.showLoading({ title: '生成订单...' });
    const orderId = await generateOrder(courseId);
    
    if (!orderId) {
      wx.hideLoading();
      return;
    }
    
    // 3. 发起支付
    wx.showLoading({ title: '正在支付...' });
    const paySuccess = await payForCourse(
      orderId, 
      enrollmentData.amount, 
      enrollmentData.courseTitle,
      courseId
    );
    
    wx.hideLoading();
    
    // 支付成功后会在 wx.requestPayment 的 success 回调中处理
    
  } catch (error) {
    wx.hideLoading();
    console.error('报名支付流程异常:', error);
    wx.showToast({ title: '操作失败，请重试', icon: 'none' });
  }
}

// 使用示例
enrollAndPay('COURSE-oABC123XYZ789-2026-01-10-a1b2c3d4');

## 📊 数据库状态变化

### 报名前
- **CourseEnrollmentTable**: 无记录
- **UserOrderTable**: 无记录
- **CoachCourseTable.current_students**: 0

### 报名后（未支付）
- **CourseEnrollmentTable**: 
  - payment_status = 1 (待支付)
  - enrollment_status = 2 (待确认)
- **UserOrderTable**: 
  - order_status = 1 (未支付)

### 支付成功后
- **CourseEnrollmentTable**:
  - payment_status = 2 (已支付)
  - enrollment_status = 1 (已报名)
  - order_id = "订单ID"
- **UserOrderTable**:
  - order_status = 2 (支付成功)
  - paymentid = "微信支付ID"
- **CoachCourseTable.current_students**: +1

## ⚠️ 注意事项

1. **Access Token 必须有效**：所有接口都需要传递有效的 access_token
2. **支付金额单位**：后端期望的金额单位是**元**（float），会自动转换为**分**（int）
3. **订单有效期**：订单默认30分钟有效期，超时需重新生成
4. **并发控制**：使用数据库事务确保支付回调的原子性
5. **重复报名检查**：已报名的课程不能重复报名（除非取消后重新报名）
6. **人数限制**：课程报名人数达到上限后不能继续报名
7. **微信支付回调**：微信官方回调由 \/notifyOrder\ 接口处理，前端还需调用 \/api/courses/payment/callback\ 更新课程报名状态
